최신 형제 서비스와 동일하게 public 스키마에 서비스 접두사 테이블(public.math_progress 외 3종)을 씁니다. 공통 로그인 모듈(cg-auth.js)의 기록 헬퍼가 public 을 기본으로 하므로 별도 설정이 필요 없습니다.

Supabase SQL 가이드

Supabase 대시보드 > SQL Editor 에서 _shared/sql/01~06 을 먼저 실행한 뒤 09-math.sql(math/supabase/migrations/001_math_schema.sql) 을 실행하면 기록 테이블·RLS·권한이 생성됩니다.

-- 수학아 놀자 기록 테이블 — 요약본. 전체는 _shared/sql/09-math.sql
create table if not exists public.math_progress (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  formula_id text not null,
  status text not null default 'done',
  learned_at timestamptz not null default now(),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (user_id, formula_id)
);
create table if not exists public.math_practice_logs (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  formula_id text, is_correct boolean not null default false,
  score int not null default 0, total int not null default 1,
  detail jsonb, created_at timestamptz not null default now()
);
create table if not exists public.math_quiz_logs (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  quiz_type text not null default 'mixed',
  score int not null, total int not null,
  detail jsonb, created_at timestamptz not null default now()
);
alter table public.math_progress enable row level security;
alter table public.math_practice_logs enable row level security;
alter table public.math_quiz_logs enable row level security;
